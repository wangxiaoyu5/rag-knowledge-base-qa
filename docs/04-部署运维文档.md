## 2. 安装部署

### 2.1 方式一：本地部署

#### 步骤 1: 下载代码

```bash
# 方式 1: 使用 git
git clone <repository-url>
cd RAG知识库问答系统

# 方式 2: 直接下载压缩包
# 解压到目标目录
cd RAG知识库问答系统
```

#### 步骤 2: 创建虚拟环境

```bash
# 创建虚拟环境
python -m venv venv

# 激活虚拟环境
# Windows:
venv\Scripts\activate

# Linux/Mac:
source venv/bin/activate

# 验证激活
which python  # 应显示 venv 路径
```

#### 步骤 3: 安装依赖

```bash
# 升级 pip
pip install --upgrade pip

# 安装基础依赖
pip install -r requirements.txt

# 验证安装
python -c "import langchain; import streamlit; print('安装成功')"
```

#### 步骤 4: 配置环境变量

```bash
# 复制环境变量模板
cp .env.example .env

# 编辑 .env 文件
# Windows:
notepad .env

# Linux/Mac:
vim .env
```

编辑内容：
```bash
OPENAI_API_KEY=your_openai_api_key_here
OPENAI_API_BASE=https://api.openai.com/v1
EMBEDDING_MODEL=BAAI/bge-large-zh
VECTOR_STORE_TYPE=faiss
```

#### 步骤 5: 准备数据

```bash
# 创建数据目录
mkdir -p data/documents
mkdir -p data/vector_db

# 复制文档到数据目录
cp /path/to/your/documents/* data/documents/
```

#### 步骤 6: 启动服务

```bash
# Web 模式
python main.py --mode web

# 或指定端口
python main.py --mode web --port 8080

# 访问 http://localhost:8501
```

### 2.2 方式二：Docker 部署

#### 构建镜像

```dockerfile
# Dockerfile
FROM python:3.10-slim

WORKDIR /app

# 安装系统依赖
RUN apt-get update && apt-get install -y \
    gcc \
    g++ \
    && rm -rf /var/lib/apt/lists/*

# 复制代码
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

COPY . .

# 创建数据目录
RUN mkdir -p data/documents data/vector_db

EXPOSE 8501

CMD ["python", "main.py", "--mode", "web", "--host", "0.0.0.0"]
```

```bash
# 构建镜像
docker build -t rag-qa-system .

# 运行容器
docker run -d \
  -p 8501:8501 \
  -v $(pwd)/data:/app/data \
  -e OPENAI_API_KEY=your_key \
  --name rag-qa \
  rag-qa-system
```

#### Docker Compose 部署

```yaml
# docker-compose.yml
version: '3.8'

services:
  rag-qa:
    build: .
    ports:
      - "8501:8501"
    volumes:
      - ./data:/app/data
      - ./models:/app/models
    environment:
      - OPENAI_API_KEY=${OPENAI_API_KEY}
      - OPENAI_API_BASE=${OPENAI_API_BASE}
      - EMBEDDING_MODEL=BAAI/bge-large-zh
      - VECTOR_STORE_TYPE=faiss
    restart: unless-stopped
    deploy:
      resources:
        limits:
          cpus: '4'
          memory: 8G
        reservations:
          cpus: '2'
          memory: 4G
```

```bash
# 启动服务
docker-compose up -d

# 查看日志
docker-compose logs -f

# 停止服务
docker-compose down
```

### 2.3 方式三：云服务器部署

#### 阿里云 ECS 部署示例

```bash
# 1. 购买 ECS 实例（推荐 4核8G）
# 2. 连接服务器
ssh root@your-server-ip

# 3. 安装 Python
apt-get update
apt-get install -y python3.10 python3-pip python3-venv

# 4. 上传代码（本地执行）
scp -r RAG知识库问答系统 root@your-server-ip:/opt/

# 5. 服务器端部署
cd /opt/RAG知识库问答系统
python3 -m venv venv
source venv/bin/activate
pip install -r requirements.txt

# 6. 配置环境变量
export OPENAI_API_KEY=your_key

# 7. 使用 systemd 管理服务
cat > /etc/systemd/system/rag-qa.service << EOF
[Unit]
Description=RAG QA System
After=network.target

[Service]
Type=simple
User=root
WorkingDirectory=/opt/RAG知识库问答系统
Environment=OPENAI_API_KEY=your_key
ExecStart=/opt/RAG知识库问答系统/venv/bin/python main.py --mode web --host 0.0.0.0 --port 8501
Restart=always

[Install]
WantedBy=multi-user.target
EOF

# 8. 启动服务
systemctl daemon-reload
systemctl enable rag-qa
systemctl start rag-qa

# 9. 查看状态
systemctl status rag-qa
journalctl -u rag-qa -f
```

## 3. 配置说明

### 3.1 环境变量配置

| 变量名 | 必填 | 默认值 | 说明 |
|--------|------|--------|------|
| OPENAI_API_KEY | 是 | - | OpenAI API 密钥 |
| OPENAI_API_BASE | 否 | https://api.openai.com/v1 | API 基础 URL |
| EMBEDDING_MODEL | 否 | BAAI/bge-large-zh | Embedding 模型 |
| EMBEDDING_DEVICE | 否 | cpu | 运行设备 (cpu/cuda) |
| VECTOR_STORE_TYPE | 否 | faiss | 向量数据库类型 |
| VECTOR_DB_PATH | 否 | ./data/vector_db | 向量数据库路径 |
| CHUNK_SIZE | 否 | 500 | 文本分块大小 |
| CHUNK_OVERLAP | 否 | 50 | 分块重叠大小 |
| LLM_MODEL | 否 | gpt-3.5-turbo | LLM 模型 |
| LLM_TEMPERATURE | 否 | 0.7 | 生成温度 |
| LOG_LEVEL | 否 | INFO | 日志级别 |

### 3.2 配置文件

#### config.yaml 详解

```yaml
# 系统配置
system:
  name: "RAG Knowledge Base QA System"
  version: "1.0.0"
  log_level: "INFO"  # DEBUG, INFO, WARNING, ERROR

# 文档处理
document:
  supported_extensions:
    - ".pdf"
    - ".docx"
    - ".txt"
    - ".md"
  chunking:
    chunk_size: 500
    chunk_overlap: 50
    strategy: "recursive"

# Embedding 配置
embedding:
  model_name: "BAAI/bge-large-zh"
  device: "cpu"  # cpu, cuda, cuda:0
  normalize_embeddings: true
  batch_size: 32

# 向量数据库
vector_store:
  type: "faiss"  # faiss, chroma
  persist_directory: "./data/vector_db"
  faiss:
    index_type: "IndexFlatIP"

# 检索配置
retrieval:
  hybrid:
    enabled: true
    vector_weight: 0.7
    bm25_weight: 0.3
  rerank:
    enabled: true
    model_name: "BAAI/bge-reranker-large"
  top_k:
    retrieve: 10
    rerank: 5

# LLM 配置
llm:
  provider: "openai"
  model: "gpt-3.5-turbo"
  temperature: 0.7
  max_tokens: 1000

# Web 配置
web:
  host: "0.0.0.0"
  port: 8501
  title: "RAG 知识库问答系统"
```

## 5. 故障排查

### 5.1 常见问题

#### 问题 1: 启动失败 - 端口被占用

**现象：**
```
OSError: [Errno 98] Address already in use
```

**解决：**
```bash
# 查找占用端口的进程
lsof -i :8501

# 或
netstat -tulpn | grep 8501

# 杀掉进程
kill -9 <PID>

# 或更换端口
python main.py --port 8502
```

#### 问题 2: 模型下载失败

**现象：**
```
HTTPError: 403 Client Error: Forbidden for url
```

**解决：**
```bash
# 设置 Hugging Face 镜像
export HF_ENDPOINT=https://hf-mirror.com

# 或使用代理
export HTTPS_PROXY=http://proxy.example.com:8080

# 手动下载模型
# 1. 从 https://huggingface.co/BAAI/bge-large-zh 下载
# 2. 放到 ./models 目录
# 3. 修改配置使用本地模型
```

#### 问题 3: 内存不足

**现象：**
```
MemoryError: Unable to allocate array
```

**解决：**
```bash
# 1. 减小批处理大小
# 修改 embedding.py 中的 batch_size

# 2. 使用更小的模型
export EMBEDDING_MODEL=BAAI/bge-base-zh

# 3. 减小分块大小
export CHUNK_SIZE=300

# 4. 增加交换空间
sudo fallocate -l 4G /swapfile
sudo chmod 600 /swapfile
sudo mkswap /swapfile
sudo swapon /swapfile
```

#### 问题 4: API 调用失败

**现象：**
```
openai.error.AuthenticationError: Incorrect API key
```

**解决：**
```bash
# 检查 API Key
echo $OPENAI_API_KEY

# 重新设置
export OPENAI_API_KEY=your_correct_key

# 检查网络
curl https://api.openai.com/v1/models \
  -H "Authorization: Bearer $OPENAI_API_KEY"

# 如果使用代理
export OPENAI_API_BASE=https://your-proxy.com/v1
```

## 7. 备份与恢复

### 7.1 数据备份

```bash
#!/bin/bash
# backup.sh

BACKUP_DIR="/backup/rag-qa/$(date +%Y%m%d)"
mkdir -p $BACKUP_DIR

# 备份向量数据库
cp -r data/vector_db $BACKUP_DIR/

# 备份原始文档
cp -r data/documents $BACKUP_DIR/

# 备份配置
cp .env $BACKUP_DIR/
cp config.yaml $BACKUP_DIR/

# 压缩
tar -czf $BACKUP_DIR.tar.gz $BACKUP_DIR
rm -rf $BACKUP_DIR

echo "备份完成: $BACKUP_DIR.tar.gz"
```

### 7.2 数据恢复

```bash
#!/bin/bash
# restore.sh

BACKUP_FILE=$1

if [ -z "$BACKUP_FILE" ]; then
    echo "用法: ./restore.sh <backup-file>"
    exit 1
fi

# 解压
tar -xzf $BACKUP_FILE

# 恢复数据
BACKUP_DIR=$(tar -tzf $BACKUP_FILE | head -1)
cp -r $BACKUP_DIR/vector_db data/
cp -r $BACKUP_DIR/documents data/
cp $BACKUP_DIR/.env .
cp $BACKUP_DIR/config.yaml .

# 清理
rm -rf $BACKUP_DIR

echo "恢复完成"
```

### 8.3 API Key 保护

```python
# 使用密钥管理服务
import boto3
from botocore.exceptions import ClientError

def get_secret():
    secret_name = "rag-qa/openai-api-key"
    region_name = "us-east-1"
    
    session = boto3.session.Session()
    client = session.client(
        service_name='secretsmanager',
        region_name=region_name
    )
    
    try:
        get_secret_value_response = client.get_secret_value(
            SecretId=secret_name
        )
        return get_secret_value_response['SecretString']
    except ClientError as e:
        raise e
```
